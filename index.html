❌ ANTES (SEM SOM)

func _on_area_2d_area_entered(area: Area2D) -> void:
    if area.name == "hurt_box_area":
        var enemy = area.get_parent()
        if enemy and enemy.has_method("take_damage"):
            
            var final_damage: int = 0
            
            if hit_simple.disabled == false:
                final_damage = 12 # Dano do Martelo
                print("Ataque detectado!")
                
            elif hit_combo.disabled == false:
                final_damage = 25 
                print("Ataque do combo detectado!")
                
            enemy.take_damage(final_damage)

✔ DEPOIS (COM SOM)

func _on_area_2d_area_entered(area: Area2D) -> void:
    if area.name == "hurt_box_area":
        $Area2D/hit.play() # genérico
        var enemy = area.get_parent()
        if enemy and enemy.has_method("take_damage"):
            
            var final_damage: int = 0
            
            if hit_simple.disabled == false:
                $Area2D/hit_simples.play() # exato
                final_damage = 12 # Martelo
                print("Ataque detectado!")
                
            elif hit_combo.disabled == false:
                $Area2D/hit_combo.play() # exato
                final_damage = 25 
                print("Ataque do combo!")
                
            enemy.take_damage(final_damage)
